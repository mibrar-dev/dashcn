```dart
/// How a label handles text that overflows its row.
enum NavigationOverflow {
  /// Clip.
  clip,
  /// Scroll the text horizontally.
  marquee,
  /// Ellipsis.
  ellipsis,
  /// Let it overflow.
  none,
}
```
