```dart
/// Position of a navigation label relative to its icon.
enum NavigationLabelPosition {
  /// Before the icon (left in LTR).
  start,
  /// After the icon (right in LTR).
  end,
  /// Above the icon.
  top,
  /// Below the icon.
  bottom,
}
```
