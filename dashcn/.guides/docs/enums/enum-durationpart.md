```dart
/// One component of a duration.
///
/// Used to order and label the fields of a duration picker; see [DatePart] for
/// the same idea applied to dates.
enum DurationPart {
  /// Whole days.
  day,
  /// Hours within the day.
  hour,
  /// Minutes within the hour.
  minute,
  /// Seconds within the minute.
  second,
}
```
