```dart
/// Position where an input feature is displayed.
enum InputFeaturePosition {
  /// Before the editable text (start side).
  leading,
  /// After the editable text (end side).
  trailing,
  /// Above the decorated field.
  above,
  /// Below the decorated field.
  below,
}
```
