```dart
/// How a surface presents itself while loading.
enum FileSurfaceLoadingMode {
  /// Swap the surface content for the loading widget.
  replace,
  /// Hide the surface entirely.
  hide,
  /// Keep the surface visible under a wrapper.
  wrap,
}
```
